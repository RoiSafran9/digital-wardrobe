# Expo Mobile App + FastAPI Backend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a personal-use Expo mobile app (capture photo → run pipeline → view results → browse closet) backed by a new FastAPI wrapper around the existing `pipeline.orchestrator.run_photo`.

**Architecture:** A new `api/` FastAPI service imports `run_photo` in-process and exposes three endpoints (`POST /runs`, `GET /items`, `GET /images/...`). A new `mobile/` Expo + TypeScript app (expo-router, 3 tabs) calls that API over the local network with plain `fetch`. No database — both the API and the closet listing read directly from the `runs/` directory the pipeline already writes.

**Tech Stack:** FastAPI, uvicorn, pytest + httpx (backend); Expo, TypeScript, expo-router, expo-image-picker, Jest + `@testing-library/react-native` (mobile).

**Spec:** `docs/superpowers/specs/2026-10-14-expo-app-design.md`

## Global Constraints

- Single-user, no auth, no multi-tenancy, no cloud deploy — phone and backend share local WiFi, app points at the Mac's LAN IP.
- Backend call style is synchronous: `POST /runs` blocks until `run_photo` completes and returns the full result in one response — no job queue, no polling.
- No database. Persistence is the existing `runs/<run_id>/<photo_name>/item_<n>/` filesystem layout; the closet listing is a directory scan, not an index.
- Mobile networking is plain `fetch`; state is local `useState`/`useEffect` — no Redux, no React Query.
- The API imports `pipeline.orchestrator.run_photo` directly (in-process) — no subprocess, no RPC layer.

## Review Focus

- `GET /images/...` with a disallowed or path-traversal filename must return `404` and never read a file outside the configured runs directory — covered in Task 1.
- `POST /runs` with a non-image/corrupt upload: `pipeline.stages.stage0_quality.run` raises `ValueError("cannot read image: ...")` for an unreadable file — the API must turn that into a `400`, not an unhandled `500` — covered in Task 3.
- `GET /items` when one item's `result.json` or `facts.json` is missing or unparseable must skip that item silently, not `500` the whole listing — covered in Task 2.
- The mobile API client (`postRun`/`getItems`) must turn a network failure (backend unreachable) into a readable thrown message, never leak a raw `TypeError` to the UI — covered in Task 6.
- The Capture screen must handle a denied permission or a cancelled picker without attempting an upload or crashing — covered in Task 8.

```python
import os
from fastapi import APIRouter
from starlette.responses import FileResponse, Response
from api.config import runs_dir

router = APIRouter()

ALLOWED_FILENAMES = {"crop.png", "generated.png"}


@router.get("/images/{run_id}/{photo_name}/{item_id}/{filename}")
def get_image(run_id: str, photo_name: str, item_id: str, filename: str) -> Response:
    if filename not in ALLOWED_FILENAMES:
        return Response(status_code=404)
    path = os.path.join(runs_dir(), run_id, photo_name, item_id, filename)
    if not os.path.isfile(path):
        return Response(status_code=404)
    return FileResponse(path, media_type="image/png")
```

```python
# api/item_loader.py
import os
from pipeline.types import ItemFacts, ItemResult


def load_item(item_dir: str) -> tuple[str, ItemResult, ItemFacts] | None:
    item_id = os.path.basename(item_dir)
    try:
        with open(os.path.join(item_dir, "result.json")) as f:
            result = ItemResult.model_validate_json(f.read())
        with open(os.path.join(item_dir, "facts.json")) as f:
            facts = ItemFacts.model_validate_json(f.read())
    except (OSError, ValueError):
        return None
    return item_id, result, facts


def image_urls(run_id: str, photo_name: str, item_id: str) -> tuple[str, str]:
    base = f"/images/{run_id}/{photo_name}/{item_id}"
    return f"{base}/crop.png", f"{base}/generated.png"
```

```python
# api/routes/items.py
import glob
import os
from fastapi import APIRouter
from api.config import runs_dir
from api.item_loader import load_item, image_urls
from api.schemas import ClosetItemOut, ItemsResponse

router = APIRouter()


@router.get("/items", response_model=ItemsResponse)
def list_items() -> ItemsResponse:
    pattern = os.path.join(runs_dir(), "*", "*", "item_*")
    item_dirs = [d for d in glob.glob(pattern) if os.path.isdir(d)]
    item_dirs.sort(key=os.path.getmtime, reverse=True)

    items: list[ClosetItemOut] = []
    for item_dir in item_dirs:
        loaded = load_item(item_dir)
        if loaded is None:
            continue
        item_id, result, facts = loaded
        photo_dir = os.path.dirname(item_dir)
        photo_name = os.path.basename(photo_dir)
        run_id = os.path.basename(os.path.dirname(photo_dir))
        crop_url, generated_url = image_urls(run_id, photo_name, item_id)
        items.append(ClosetItemOut(
            run_id=run_id, item_id=item_id, tier=result.tier, facts=facts,
            crop_url=crop_url, generated_url=generated_url,
        ))
    return ItemsResponse(items=items)
```

```python
# api/routes/runs.py
import glob
import os
import shutil
import tempfile
import time
from typing import Callable
from fastapi import APIRouter, Depends, File, UploadFile
from fastapi.responses import JSONResponse
from api.config import runs_dir
from api.item_loader import load_item, image_urls
from api.schemas import RunResponse, RunItemOut, QualityOut
from cli import build_providers
from pipeline.orchestrator import run_photo, Providers
from pipeline.run_log import RunLogger

router = APIRouter()

ProviderFactory = Callable[[RunLogger], Providers]


def default_provider_factory(logger: RunLogger) -> Providers:
    return build_providers(logger)


def get_provider_factory() -> ProviderFactory:
    return default_provider_factory


@router.post("/runs", response_model=RunResponse)
async def create_run(
    photo: UploadFile | None = File(None),
    provider_factory: ProviderFactory = Depends(get_provider_factory),
):
    if photo is None or not photo.filename:
        return JSONResponse(status_code=400, content={"error": "no photo provided"})

    run_id = time.strftime("%Y%m%d-%H%M%S-") + str(int(time.time() * 1000) % 1000)
    photo_name = os.path.splitext(photo.filename)[0] or "photo"
    run_root = os.path.join(runs_dir(), run_id)
    output_dir = os.path.join(run_root, photo_name)
    os.makedirs(run_root, exist_ok=True)

    suffix = os.path.splitext(photo.filename)[1] or ".jpg"
    with tempfile.NamedTemporaryFile(suffix=suffix, delete=False) as tmp:
        shutil.copyfileobj(photo.file, tmp)
        photo_path = tmp.name

    logger = RunLogger(os.path.join(run_root, "run_log.jsonl"))
    providers = provider_factory(logger)

    try:
        result = await run_photo(photo_path, output_dir, providers, logger)
    except ValueError as e:
        return JSONResponse(status_code=400, content={"error": str(e)})
    except Exception as e:
        return JSONResponse(status_code=500, content={"error": str(e)})

    item_dirs = sorted(
        (d for d in glob.glob(os.path.join(output_dir, "item_*")) if os.path.isdir(d)),
        key=lambda d: int(os.path.basename(d).split("_")[1]),
    )
    items_out: list[RunItemOut] = []
    for item_dir in item_dirs:
        loaded = load_item(item_dir)
        if loaded is None:
            continue
        item_id, item_result, facts = loaded
        crop_url, generated_url = image_urls(run_id, photo_name, item_id)
        items_out.append(RunItemOut(
            item_id=item_id, tier=item_result.tier, reason=item_result.reason,
            retake_instruction=item_result.retake_instruction, facts=facts,
            crop_url=crop_url, generated_url=generated_url,
        ))

    return RunResponse(
        run_id=run_id,
        quality=QualityOut(status=result["quality"].status, reason=result["quality"].reason),
        items=items_out,
        failed=result["failed"],
    )
```

```ts
// mobile/src/api/client.ts
import type { RunResponse, ItemsResponse, ApiError } from "./types";

export const API_BASE_URL = process.env.EXPO_PUBLIC_API_URL ?? "http://192.168.1.238:8000";

async function parseJsonOrThrow<T>(response: Response): Promise<T> {
  const body = await response.json();
  if (!response.ok) {
    const message = (body as ApiError).error ?? `request failed with status ${response.status}`;
    throw new Error(message);
  }
  return body as T;
}

export async function postRun(photoUri: string): Promise<RunResponse> {
  const formData = new FormData();
  formData.append("photo", {
    uri: photoUri,
    name: "photo.jpg",
    type: "image/jpeg",
  } as any);

  let response: Response;
  try {
    response = await fetch(`${API_BASE_URL}/runs`, { method: "POST", body: formData });
  } catch {
    throw new Error("can't reach server — check WiFi/IP");
  }
  return parseJsonOrThrow<RunResponse>(response);
}

export async function getItems(): Promise<ItemsResponse> {
  let response: Response;
  try {
    response = await fetch(`${API_BASE_URL}/items`);
  } catch {
    throw new Error("can't reach server — check WiFi/IP");
  }
  return parseJsonOrThrow<ItemsResponse>(response);
}
```

```tsx
  const { setLastRun } = useLastRun();
  const router = useRouter();

  async function pickAndUpload(fromCamera: boolean) {
    setError(null);
    const permission = fromCamera
      ? await ImagePicker.requestCameraPermissionsAsync()
      : await ImagePicker.requestMediaLibraryPermissionsAsync();
    if (!permission.granted) {
      setError("permission denied — enable camera/photos access in Settings");
      return;
    }

    const result = fromCamera
      ? await ImagePicker.launchCameraAsync({ quality: 0.8 })
      : await ImagePicker.launchImageLibraryAsync({ quality: 0.8 });

    if (result.canceled || result.assets.length === 0) {
      return;
    }

    setLoading(true);
    try {
      const run = await postRun(result.assets[0].uri);
      setLastRun(run);
      router.push("/results");
    } catch (e) {
      setError(e instanceof Error ? e.message : "upload failed");
    } finally {
      setLoading(false);
    }
  }

  return (
    <View style={styles.container}>
      <Button title="Take Photo" onPress={() => pickAndUpload(true)} disabled={loading} />
      <Button title="Pick From Library" onPress={() => pickAndUpload(false)} disabled={loading} />
      {loading ? <ActivityIndicator testID="upload-spinner" /> : null}
      {error ? <Text style={styles.error}>{error}</Text> : null}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: "center", alignItems: "center", gap: 12 },
  error: { color: "#c62828" },
});
```

```tsx
// mobile/app/closet.tsx
import React, { useCallback, useEffect, useState } from "react";
import { FlatList, Image, Pressable, RefreshControl, Modal, View } from "react-native";
import { getItems } from "../src/api/client";
import { ItemCard } from "../src/components/ItemCard";
import type { ClosetItem } from "../src/api/types";

export default function ClosetScreen() {
  const [items, setItems] = useState<ClosetItem[]>([]);
  const [refreshing, setRefreshing] = useState(false);
  const [selected, setSelected] = useState<ClosetItem | null>(null);

  const load = useCallback(async () => {
    const response = await getItems();
    setItems(response.items);
  }, []);

  useEffect(() => {
    load();
  }, [load]);

  async function onRefresh() {
    setRefreshing(true);
    await load();
    setRefreshing(false);
  }

  return (
    <View style={{ flex: 1 }}>
      <FlatList
        data={items}
        numColumns={3}
        keyExtractor={(item) => `${item.run_id}-${item.item_id}`}
        refreshControl={<RefreshControl refreshing={refreshing} onRefresh={onRefresh} />}
        renderItem={({ item }) => (
          <Pressable testID={`thumb-${item.run_id}-${item.item_id}`} onPress={() => setSelected(item)}>
            <Image source={{ uri: item.generated_url }} style={{ width: 100, height: 100 }} />
          </Pressable>
        )}
      />
      <Modal visible={selected !== null} onRequestClose={() => setSelected(null)}>
        {selected ? <ItemCard item={selected} /> : null}
      </Modal>
    </View>
  );
}
```
