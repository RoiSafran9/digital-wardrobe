# 360 "On Body" Turn View Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An owner taps "View in 360" on a closet item; the server generates 4 ghost-mannequin angles in the
background and the item sheet shows them as a draggable turn view labelled "360 · AI view".

**Architecture:** A new pipeline step (`pipeline/turn.py`) asks the image provider for 4 views (front first, then
right/back/left anchored on the generated front) and cuts each out locally. A new API router starts it as a
FastAPI background task and records progress in `turn/status.json`; the closet listing exposes the result. The
app adds a `TurnView` component, a `useTurn` polling hook, a third "360" segment in the item sheet, and a closet badge.

**Tech Stack:** Python 3 / FastAPI / google-genai (async client `client.aio`) / pytest + pytest-asyncio;
Expo SDK 57 / React Native 0.86 / Jest + @testing-library/react-native. No new packages.

**Spec:** `docs/superpowers/specs/2026-10-17-turn-360-design.md`

## Global Constraints

- Optional and explicit: generation only on a "View in 360" press; never automatic (~16¢, ~1 min).
- Views in turn order: `front`, `right`, `back`, `left`; files `<view>.png` and `<view>_cutout.png` in `runs/<run>/<photo>/<item>/turn/`.
- `status.json` = `{"state": "running" | "done" | "failed", "error": string | null, "started_at": iso}`.
- A `running` status older than 5 minutes reports `failed` with error `"Interrupted — try again"`.
- POST while running → `409`; item without generated front → `404`; POST returns `202 {state: "running"}`.
- The UI labels the 360 "360 · AI view".
- Image model `gemini-2.5-flash-image` (existing `MODEL_IMAGE`); no new paid services.
- The generation call must not block the event loop (use `client.aio`), because the app polls during the job. Cutout runs inline (never `to_thread`: it segfaulted before).
- Native iOS look: `colors`/`type`/`radius` tokens from `mobile/src/theme.ts`; SF Symbols via `Icon`.
- Reduce Motion: no scrub crossfade; tap/swipe switch views with a short fade.
- Regenerating moves the previous `turn/` to `runs/.trash/...` (removal stays recoverable).

## Review Focus

1. Server restarted mid-job → status stuck `running`: must read as failed after 5 min and allow retry (Task 3 test `test_stale_running_job_reads_as_failed_and_can_restart`).
2. Item without a back photo → back view must not invent prints; provider omits the back reference and adds the no-back instruction (Task 1 test `test_generate_view_without_back_says_so`).
3. Sheet closed during a job, reopened later → polling resumes from the item's `running` turn (Task 6 test `resumes polling when the item is already running`).
4. Horizontal drag on the turn view inside the horizontally paging item pager → the turn view must own the gesture (Task 5: `onMoveShouldSetPanResponderCapture` + `onPanResponderTerminationRequest: () => false`; checked manually on device).
5. A view whose cutout failed → the view still shows (falls back to the white-background image via `GarmentImage`) (Task 3 test `test_done_lists_views_and_tolerates_a_missing_cutout`).

```python
"""The 360 "on body" view: four ghost-mannequin angles of an item, generated from its catalog images.

The front is generated first and then passed back as an anchor, so the other angles keep its volume
and proportions. Each view is cut out locally like every other garment image.
"""
import os
from pipeline.stages import cutout

TURN_ORDER = ("front", "right", "back", "left")


class TurnFailed(Exception):
    """The 360 view couldn't be made; the message is shown to the user."""


async def run_turn(front_path: str, back_path: str | None, out_dir: str, imagegen) -> list[str]:
    os.makedirs(out_dir, exist_ok=True)
    paths: list[str] = []
    anchor: str | None = None
    for view in TURN_ORDER:
        try:
            data = await imagegen.generate_view(front_path, back_path, view, anchor)
        except Exception as e:
            raise TurnFailed("couldn't create the 360 view — try again") from e
        path = os.path.join(out_dir, f"{view}.png")
        with open(path, "wb") as f:
            f.write(data)
        if view == "front":
            anchor = path
        cutout.run(path, os.path.join(out_dir, f"{view}_cutout.png"))
        paths.append(path)
    return paths
```

```python
import json
from datetime import datetime, timezone

# A job still "running" after this long died with the server; it can be started again.
STALE_AFTER_SECONDS = 5 * 60
INTERRUPTED = "Interrupted — try again"


def write_turn_status(turn_dir: str, state: str, started_at: str, error: str | None = None) -> None:
    with open(os.path.join(turn_dir, "status.json"), "w") as f:
        json.dump({"state": state, "error": error, "started_at": started_at}, f)


def load_turn(item_dir: str, run_id: str, photo_name: str, item_id: str):
    """The item's 360 view as a TurnOut, or None when one was never started."""
    from api.schemas import TurnOut, TurnViewOut
    from pipeline.turn import TURN_ORDER
    turn_dir = os.path.join(item_dir, "turn")
    try:
        with open(os.path.join(turn_dir, "status.json")) as f:
            status = json.load(f)
        state, error = status["state"], status.get("error")
        started = datetime.fromisoformat(status["started_at"])
    except (OSError, ValueError, KeyError, TypeError):
        return None
    if state == "running" and (datetime.now(timezone.utc) - started).total_seconds() > STALE_AFTER_SECONDS:
        return TurnOut(state="failed", error=INTERRUPTED)
    if state != "done":
```

```python
):
    item_dir = _item_dir(run_id, photo_name, item_id)
    front = os.path.join(item_dir, "generated.png") if item_dir else None
    if not front or not os.path.exists(front):

    current = load_turn(item_dir, run_id, photo_name, item_id)

    turn_dir = os.path.join(item_dir, "turn")
    if os.path.exists(turn_dir):
        # Regenerating never loses the previous views: they go to the trash like any removal.
        trash = os.path.join(runs_dir(), TRASH_DIR, run_id, photo_name, item_id, f"turn-{int(time.time())}")
        os.makedirs(os.path.dirname(trash), exist_ok=True)
        shutil.move(turn_dir, trash)
    os.makedirs(turn_dir)
    started_at = datetime.now(timezone.utc).isoformat()
    write_turn_status(turn_dir, "running", started_at)

    back = os.path.join(item_dir, "back", "generated.png")
    imagegen = provider_factory(RunLogger(os.path.join(item_dir, "turn_log.jsonl"))).imagegen
    background.add_task(_make_turn, front, back if os.path.exists(back) else None, turn_dir, started_at, imagegen)
```

`mobile/src/turn.ts`:

```ts
// Turning a garment by dragging: one view per STEP_PX of horizontal drag, on a closed circle.
export const STEP_PX = 90;

export function wrap(i: number, n: number): number {
  return ((i % n) + n) % n;
}

// Dragging left (negative dx) turns the garment toward the next view.
export function dragPosition(start: number, dx: number): number {
  return start - dx / STEP_PX;
}

export function snap(pos: number, n: number): number {
  return wrap(Math.round(pos), n);
}

// Distance around the circle between a view and the current position; fully visible at 0,
// gone at 1 — so the two neighbours of the position crossfade.
export function opacityFor(view: number, pos: number, n: number): number {
  const d = Math.abs(wrap(pos, n) - view);
  return Math.max(0, 1 - Math.min(d, n - d));
}
```

```tsx
import React, { useMemo, useRef, useState } from "react";
import { PanResponder, Pressable, Text, View } from "react-native";
import { GarmentImage } from "./GarmentImage";
import { colors, type } from "../theme";
import { useReducedMotion } from "../useReducedMotion";
import { Rise } from "../motion/Rise";
import { dragPosition, opacityFor, snap, wrap } from "../turn";
import { haptics } from "../haptics";

export type TurnFrame = { label: string; path: string; cutoutPath?: string | null };

type Props = { views: TurnFrame[]; label?: string };

// The garment on an invisible body: drag sideways to turn it, tap to step one view.
// Every view stays mounted (stacked) so turning is instant; drag progress crossfades neighbours.
export function TurnView({ views, label }: Props) {
  const n = views.length;
  const reduced = useReducedMotion();
  const [index, setIndex] = useState(0);
  const [pos, setPos] = useState(0);          // fractional position while dragging
  const dragging = useRef(false);
  const start = useRef(0);

  const go = (next: number) => {
    const i = wrap(next, n);
    if (i !== index) haptics.selection();
    setIndex(i);
    setPos(i);
  };

  const pan = useMemo(
    () =>
      PanResponder.create({
        // Claim clearly horizontal drags before the item pager does.
        onMoveShouldSetPanResponderCapture: (_, g) => Math.abs(g.dx) > 8 && Math.abs(g.dx) > Math.abs(g.dy),
        onPanResponderTerminationRequest: () => false,
        onPanResponderGrant: () => {
          dragging.current = true;
          start.current = index;
        },
        onPanResponderMove: (_, g) => {
          if (!reduced) setPos(dragPosition(start.current, g.dx));
        },
        onPanResponderRelease: (_, g) => {
          dragging.current = false;
          // Reduce Motion: no scrub; a swipe steps one view with a short fade.
          go(reduced ? start.current + (g.dx < 0 ? 1 : -1) : snap(dragPosition(start.current, g.dx), n));
        },
      }),
    [index, n, reduced],
  );

  return (
    <View>
      <Pressable
        {...pan.panHandlers}
        accessible
        accessibilityRole="adjustable"
        accessibilityLabel="360 view"
        accessibilityHint="Swipe up or down to turn the garment"
        accessibilityValue={{ text: views[index].label }}
        accessibilityActions={[{ name: "increment" }, { name: "decrement" }]}
        onAccessibilityAction={(e) => go(index + (e.nativeEvent.actionName === "increment" ? 1 : -1))}
        onPress={() => go(index + 1)}
        style={{ width: "100%", aspectRatio: 1 }}
      >
        {reduced ? (
          <Rise key={index} distance={0} style={{ flex: 1 }}>
            <GarmentImage path={views[index].path} cutoutPath={views[index].cutoutPath} style={{ width: "100%", height: "100%" }} />
          </Rise>
        ) : (
          views.map((v, i) => (
            <View key={v.path} pointerEvents="none" style={{ position: "absolute", top: 0, left: 0, right: 0, bottom: 0, opacity: opacityFor(i, pos, n) }}>
              <GarmentImage path={v.path} cutoutPath={v.cutoutPath} style={{ width: "100%", height: "100%" }} />
            </View>
          ))
        )}
      </Pressable>
      <View style={{ flexDirection: "row", justifyContent: "center", alignItems: "center", gap: 6, marginTop: 10 }}>
        {views.map((v, i) => (
          <View
            key={v.path}
            testID={`turn-dot-${i}`}
            style={{ width: 6, height: 6, borderRadius: 3, backgroundColor: i === index ? colors.label : colors.separatorStrong }}
          />
        ))}
      </View>
      {label ? (
        <Text style={{ ...type.caption, color: colors.secondaryLabel, textAlign: "center", marginTop: 6 }}>{label}</Text>
      ) : null}
    </View>
  );
}
```

```ts
// Starts the 360 job and polls it while the sheet is open; progress lives on the item
// (via onUpdated), so closing and reopening the sheet resumes where it was.
export function useTurn(item: ClosetItem | null, onUpdated?: (item: ClosetItem) => void) {
  const [error, setError] = useState<string | null>(null);
  const turn = item?.turn ?? null;
  const latest = useRef({ item, onUpdated });
  latest.current = { item, onUpdated };

  useEffect(() => {
    if (!item || turn?.state !== "running") return;
    let alive = true;
    const timer = setInterval(async () => {
      try {
        const next = await getTurn(item);
        const { item: current, onUpdated: report } = latest.current;
        if (alive && current && next.state !== "running") report?.({ ...current, turn: next });
      } catch {
        // A missed poll is fine; the next one tries again.
      }
    }, POLL_MS);
    return () => {
      alive = false;
      clearInterval(timer);
    };
  }, [item?.run_id, item?.photo_name, item?.item_id, turn?.state]);

  async function start() {
    if (!item || !onUpdated) return;
    setError(null);
    try {
      onUpdated({ ...item, turn: await startTurn(item) });
    } catch (e) {
      setError(e instanceof Error ? e.message : "Couldn't start the 360 view.");
    }
  }

  return { start, turn, error };
}
```
